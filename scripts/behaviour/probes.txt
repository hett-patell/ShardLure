# The bot family that scp-uploads its own ELF and runs it as a liveness test
# (exec-emulation). Cowrie must never run it: a real box runs the matching-arch
# binary silently, rc 0. Upload and run share one channel here ("#harness:
# scp-stdin"); the cross-channel cases below are the bot's real shape.
scp-upload-run	scp -t /root/x >/dev/null; ./x
# echo -e "new\nnew" | passwd: unpatched v3.1.1 waits on stdin forever.
passwd-stdin	echo -e "p\np" | passwd
# The same bot's real shape (Task 3b): `scp -t <path>` on one channel, then run
# <path> on the next channel of the same connection. Stock v3.1.1 fails both
# halves: each channel gets a fresh fake filesystem (connection-shared-fs.py),
# and the upload is saved under the C-record's name (hw) instead of the -t
# path (scp-sink-target.py). A directory target
# keeps the record's name, as OpenSSH's sink does. Nothing is ever executed:
# exec-emulation answers the run (rc 0, no output).
scp-cross-channel	ls -l /tmp/x; /tmp/x; echo rc=$?
scp-cross-channel-dir	ls -l /tmp/hw; /tmp/hw; echo rc=$?
# A file written by a redirection on one channel, read and appended on the
# next. With one fake filesystem per connection, the channel close that
# finalises the redirection must leave the node readable (connection-shared-fs
# re-points it; it used to hang the session with FileNotFoundError).
redirect-cross-channel	cat /tmp/m; echo more >> /tmp/m; cat /tmp/m
# Concurrent channels on one connection (Task 3b review). I-1: a redirection
# into a file whose SFTP upload is still open must neither truncate nor
# rename the upload's temp file; the upload must still close and be captured
# byte for byte (needs --downloads-dir). I-2: /proc/uptime must keep working
# on a channel after a later-opened channel closes (the persona value itself
# is proc-uptime-advances' job).
sftp-inflight-redirect	echo X > /tmp/up; echo rc=$?
proc-uptime-overlap	sleep 2; cat /proc/uptime | wc -w
# grep options (Phase B Task 4). v3.1.1's grep parses -i -q -c -v -l -o and
# ignores them: the profiler's `lspci | grep -i vga` never matches (GPU field

# `grep -c ^processor` cpus fallback prints lines instead of a count.
grep-i-vga	printf 'VGA x\n' | grep -i vga
grep-q-lm	grep -q "lm" /proc/cpuinfo && echo x86_64
grep-q-no-match	grep -q zzz_no_match /proc/cpuinfo
grep-c-processor	grep -c "^processor" /proc/cpuinfo
grep-c-no-match	printf 'a\n' | grep -c zzz
grep-v	printf 'a\nb\nc\n' | grep -v b

# grep pattern dialects and --color (Task 4 review I1/I2): GNU's default BRE
# makes \| \( \{ \+ \? the operators (`grep 'model name\|Hardware'` is a recon
# idiom), -E is ERE, -F is literal; stock compiled every pattern as Python re
# and refused --color=auto.
grep-bre-alternation	grep 'model name\|Hardware' /proc/cpuinfo
grep-ere-alternation	printf 'a\nb\nc\n' | grep -E 'a|b'
grep-bre-plus-literal	printf 'a+\naa\n' | grep 'a+'
grep-fixed-string	printf 'a.b\naxb\n' | grep -F 'a.b'
grep-bre-interval	printf '1 23\n4 5\n' | grep '[[:digit:]]\{2\}'
grep-bre-unmatched	grep '\(' /proc/cpuinfo
grep-color	printf 'foo\n' | grep --color=auto foo; printf 'bar\n' | grep --colour bar
